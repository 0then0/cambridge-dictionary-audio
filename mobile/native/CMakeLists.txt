cmake_minimum_required(VERSION 3.22)
project(cambridge_phrase C)
set(LAME_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../build/mobile-codec/android-${ANDROID_ABI}")
set(LAME_LIBRARY "${LAME_ROOT}/lib/libmp3lame.a")
set(LAME_INCLUDE "${LAME_ROOT}/include/lame")
if(NOT EXISTS "${LAME_LIBRARY}")
  message(FATAL_ERROR "Run scripts/build_mobile_codec.py android-${ANDROID_ABI} first")
endif()
add_library(mp3lame STATIC IMPORTED)
set_target_properties(mp3lame PROPERTIES IMPORTED_LOCATION "${LAME_LIBRARY}")
add_library(cambridge_phrase SHARED phrase_codec.c android_jni.c)
target_include_directories(cambridge_phrase PRIVATE "${LAME_INCLUDE}")
target_link_libraries(cambridge_phrase mp3lame m)
target_compile_options(cambridge_phrase PRIVATE -Wall -Wextra -Werror)
target_link_options(cambridge_phrase PRIVATE "-Wl,-z,max-page-size=16384")
